import assert from 'node:assert/strict';
import { readFileSync,writeFileSync,mkdtempSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { pathToFileURL,fileURLToPath } from 'node:url';
const here=fileURLToPath(new URL('.',import.meta.url));
const dir=mkdtempSync(join(tmpdir(),'interny-matching-'));
writeFileSync(join(dir,'matching.mjs'),readFileSync(join(here,'matching.js'),'utf8').replace("'./cityCoords.js'","'./cityCoords.mjs'"));
writeFileSync(join(dir,'cityCoords.mjs'),readFileSync(join(here,'cityCoords.js'),'utf8'));
writeFileSync(join(dir,'onboarding.mjs'),readFileSync(join(here,'onboarding.js'),'utf8'));
const M=await import(pathToFileURL(join(dir,'matching.mjs')).href);
const O=await import(pathToFileURL(join(dir,'onboarding.mjs')).href);
const user={grade:'10th Grade',interests:['Computer Science'],state:'VA',city:'Richmond',location:'Richmond, VA',locationConfirmed:true,locationSource:'home',logisticsAnswered:true,formatPreference:'remote',travelWillingness:'local',maxCommuteMiles:25};
const item={id:'test',field:'Computer Science',tags:['Computer Science'],gradesShort:'9-12',payType:'Stipend',remote:true};
let passed=0;function test(name,fn){try{fn();passed++;console.log('PASS '+name);}catch(e){console.error('FAIL '+name);throw e;}}
const breakdown=(i={},u={})=>M.computeMatchBreakdown({...item,...i},{...user,...u});
test('Only grade cannot create a numerical fit',()=>assert.equal(M.computeMatchScore(item,{grade:'10th Grade'}),null));
test('Only priorities cannot create a numerical fit',()=>assert.equal(M.computeMatchScore(item,{priorities:{interests:3}}),null));
test('Exploring is not a made-up interest score',()=>assert.equal(breakdown({}, {interests:[],exploring:true}).total,null));
test('Unconfirmed home prevents a numerical fit',()=>assert.equal(breakdown({}, {locationConfirmed:false}).total,null));
test('Featured status never changes personal fit',()=>assert.equal(breakdown({featured:true}).total,breakdown({featured:false}).total));
test('School mention never changes personal fit',()=>assert.equal(breakdown({requirements:'Richmond Public School'}, {school:{name:'Richmond Public School'}}).total,breakdown().total));
test('Narrow qualifying grade ranges do not earn a bonus',()=>assert.equal(breakdown({gradesShort:'10'}).total,breakdown().total));
test('Missing birthday is unknown on age-restricted program',()=>assert.equal(breakdown({minAge:16}).eligibility.unknown,true));
test('Missing birthday warning reaches UI breakdown',()=>assert.ok(breakdown({minAge:16}).checksNeeded.some(x=>x.includes('birthday'))));
test('Known age at explicit cutoff rejects under-age student',()=>assert.equal(breakdown({minAge:16,ageAsOfDate:'2026-09-05'}, {birthday:'2012-01-01'}).ineligible,true));
test('Birthday at future cutoff can qualify',()=>assert.equal(breakdown({minAge:16,ageAsOfDate:'2027-06-01'}, {birthday:'2011-05-01'}).ineligible,false));
test('Unknown cutoff cannot prove future age rejection',()=>assert.equal(breakdown({minAge:16}, {birthday:'2012-01-01'}).eligibility.unknown,true));
test('Missing gender produces a confirmation label',()=>assert.equal(M.getMatchLabel(breakdown({requiredGender:'female'})),'Needs confirmation'));
test('Known gender mismatch is explicitly rejected',()=>assert.equal(breakdown({requiredGender:'female'},{gender:'male'}).ineligible,true));
test('Missing race produces a confirmation check',()=>assert.equal(breakdown({requiredRace:'Black'}).eligibility.unknown,true));
test('Known grade mismatch is rejected',()=>assert.equal(breakdown({gradesShort:'11-12'}).ineligible,true));
test('Age text cannot be interpreted as a grade',()=>assert.equal(breakdown({gradesShort:'Ages 10-18'},{grade:'11th Grade',age:16}).ineligible,false));
test('Age text with unknown grade remains a check',()=>assert.equal(breakdown({gradesShort:'Ages 10-18'}).eligibility.unknown,true));
test('Invalid birthdays are rejected',()=>assert.equal(M.ageFromBirthday('2010-02-31'),null));
test('Birthday parsing handles leap dates',()=>assert.equal(M.ageFromBirthday('2012-02-29','2026-09-05'),14));
test('Future birthday is rejected',()=>assert.equal(M.ageFromBirthday('2099-01-01'),null));
test('Unknown compensation is never unpaid/free',()=>assert.equal(M.detectPayType({...item,payType:null}).isUnpaid,false));
test('Unknown compensation is explicitly unknown',()=>assert.equal(M.detectPayType({...item,payType:null}).isUnknown,true));
test('Program fees do not satisfy required pay',()=>assert.equal(breakdown({payType:'Paid'},{paidRequired:true}).practical.status,'unmet'));
test('Unknown pay requires confirmation for required pay',()=>assert.equal(breakdown({payType:null},{paidRequired:true}).practical.status,'unknown'));
test('Stipend satisfies required student pay',()=>assert.equal(breakdown({}, {paidRequired:true}).practical.status,'viable'));
test('Remote-only excludes nationwide in-person work',()=>assert.equal(M.isLocationEligible({remote:false,location:'Various locations nationwide'},{remoteOnly:true}),false));
test('Remote-only excludes hybrid recommendations',()=>assert.equal(M.isRecommendable({...item,remote:false,location:'Hybrid, Richmond, VA'},{...user,remoteOnly:true}),false));
test('Same-state travel is not a local commute',()=>assert.equal(breakdown({remote:false,location:'San Francisco, CA'},{location:'San Diego, CA',state:'CA',city:'San Diego'}).practical.status,'unmet'));
test('Interest priority cannot override a commute limit',()=>assert.equal(M.isRecommendable({...item,remote:false,location:'Los Angeles, CA'},{...user,priorities:{interests:5}}),false));
test('Explicit Maine state overrides Portland city ambiguity',()=>assert.equal(breakdown({remote:false,location:'Portland, ME'},{location:'Eugene, OR',state:'OR',travelWillingness:'state'}).practical.status,'unmet'));
test('Unknown site stays uncertain instead of feasible',()=>assert.equal(breakdown({remote:false,location:'Site to be announced'}).practical.status,'unknown'));
test('School state does not satisfy home residency',()=>assert.equal(breakdown({locationEligibility:[{type:'state',value:'CA',scope:'residence'}]},{school:{state:'CA',city:'Los Angeles'}}).ineligible,true));
test('School attendance restriction uses school evidence',()=>assert.equal(breakdown({locationEligibility:[{type:'state',value:'CA',scope:'school'}]},{school:{state:'CA',city:'Los Angeles'}}).ineligible,false));
test('Legacy school-derived home requires confirmation',()=>assert.equal(breakdown({locationEligibility:[{type:'state',value:'VA'}]},{locationConfirmed:false,locationSource:'school',school:{state:'VA',city:'Richmond'}}).eligibility.unknown,true));
test('Dates outside availability are an explicit limit',()=>assert.equal(breakdown({startDate:'2027-06-01',endDate:'2027-08-01'},{availableFrom:'2027-07-01',availableUntil:'2027-08-30'}).practical.status,'unmet'));
test('Unavailable dates prompt a check',()=>assert.equal(breakdown({}, {availableFrom:'2027-07-01'}).practical.status,'unknown'));
test('Closed programs never become recommendations',()=>assert.equal(M.isRecommendable({...item,deadlineDate:'2020-01-01'},user),false));
test('Unknown eligibility cannot trigger high-match alerts',()=>assert.equal(M.isRecommendable({...item,requiredGender:'female'},user,{confirmedOnly:true}),false));
test('Confirmed remote program can be recommended',()=>assert.equal(M.isRecommendable(item,user,{confirmedOnly:true}),true));
test('Readiness tracks actual requested materials',()=>assert.deepEqual(breakdown({requirements:'Upload a resume and transcript.'},{readiness:['hasResume','hasEssay']}).readiness.tasks.map(t=>[t.key,t.ready]),[['hasResume',true],['hasTranscript',false]]));
test('Readiness and GPA cannot alter preference score',()=>assert.equal(breakdown({}, {gpaRange:'high',readiness:['hasResume','hasEssay','hasTranscript']}).total,breakdown().total));
test('Selectivity preference behaves independently from GPA',()=>assert.ok(breakdown({competitiveness:'competitive'},{gpaRange:'below',selectivityPreference:'competitive'}).total>breakdown({competitiveness:'open'},{gpaRange:'below',selectivityPreference:'competitive'}).total));
test('Profile updates immediately change pure scoring result',()=>assert.ok(breakdown({}, {formatPreference:'remote'}).total>breakdown({}, {formatPreference:'inperson'}).total));
test('Manual state validation rejects ZZ',()=>assert.equal(M.isValidState('ZZ'),false));
test('Manual state validation accepts DC',()=>assert.equal(M.isValidState('DC'),true));
test('Re-entering onboarding preserves existing answers',()=>assert.equal(O.initialProfile({...user,name:'Existing'}).name,'Existing'));
test('Draft merges instead of blanking existing answers',()=>assert.deepEqual(O.mergeOnboardingDraft(user,{profile:{grade:'11th Grade'}}).interests,user.interests));
test('Profile completeness does not use GPA',()=>assert.equal(M.getProfileCoverage(user).percent,M.getProfileCoverage({...user,gpaRange:'high'}).percent));
test('Known mismatch ranks below unresolved eligibility',()=>assert.ok(M.getRecommendationTier({...item,gradesShort:'12'},user)>M.getRecommendationTier({...item,requiredGender:'female'},user)));
console.log(`\n${passed} behavior checks passed.`);
